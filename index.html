import os import gradio as gr from fontTools.ttLib import TTFont, newTable from fontTools.pens.cu2quPen import Cu2QuPen from fontTools.pens.ttGlyphPen import TTGlyphPen import time import zipfile import io def otf_to_ttf_segments(font): if 'CFF ' not in font: return font glyph_set = font.getGlyphSet() glyf = newTable('glyf') glyf.glyphs = {} glyph_order = font.getGlyphOrder() for name in glyph_order: old_glyph = glyph_set[name] tt_pen = TTGlyphPen(glyph_set) cu2qu_pen = Cu2QuPen(tt_pen, max_err=1.0, reverse_direction=True) old_glyph.draw(cu2qu_pen) glyf.glyphs[name] = tt_pen.glyph() font['glyf'] = glyf font['loca'] = newTable('loca') if 'CFF ' in font: del font['CFF '] if 'VORG' in font: del font['VORG'] font.sfntVersion = "\x00\x01\x00\x00" tmp = io.BytesIO() font.save(tmp) tmp.seek(0) return TTFont(tmp) def process_multiple_fonts_no_rename(input_files, sb=30): if not input_files: return None, "ファイルを選択してください" timestamp = int(time.time()) output_dir = f"processed_{timestamp}" os.makedirs(output_dir, exist_ok=True) processed_paths = [] log_messages = [] for file_obj in input_files: input_path = file_obj.name filename = os.path.basename(input_path) # 形式をTTFに統一 output_filename = f"fixed_{os.path.splitext(filename)[0]}.ttf" output_path = os.path.join(output_dir, output_filename) try: font = TTFont(input_path) # 1. 編集のために一時的にTTF形式のデータ構造にする # (OTFの場合でも、内部のglyfテーブルを操作するため) if 'CFF ' in font: font = otf_to_ttf_segments(font) hmtx = font['hmtx'] glyf = font['glyf'] x_mins, x_maxs, y_mins, y_maxs = [], [], [], [] max_advance = 0 # 2. 左右余白の調整 for name in font.getGlyphOrder(): # 【変更点】全角空白(uni3000)と半角空白(space)はスキップ if name in ['space', 'uni3000']: continue glyph = glyf[name] glyph.recalcBounds(glyf) if hasattr(glyph, 'xMin'): orig_width = glyph.xMax - glyph.xMin dx = sb - glyph.xMin if glyph.isComposite(): for component in glyph.components: component.x = int(component.x + dx) elif hasattr(glyph, 'coordinates'): glyph.coordinates.translate((dx, 0)) glyph.recalcBounds(glyf) new_advance = int(sb + orig_width + sb) hmtx[name] = [new_advance, int(sb)] x_mins.append(glyph.xMin); x_maxs.append(glyph.xMax) y_mins.append(glyph.yMin); y_maxs.append(glyph.yMax) if new_advance > max_advance: max_advance = new_advance else: hmtx[name] = [int(sb * 2), int(sb)] # 3. メタデータ同期 if x_mins: font['head'].xMin, font['head'].yMin = min(x_mins), min(y_mins) font['head'].xMax, font['head'].yMax = max(x_maxs), max(y_maxs) font['hhea'].advanceWidthMax = max_advance # 4. 【変更点】Subsetを通さず、不具合の出やすいテーブルだけ削除して直接保存 # これにより「too much data」エラーを回避します bad_tables = ['GSUB', 'GPOS', 'DSIG', 'hdmx', 'LTSH', 'PCLT', 'VDMX'] for t in bad_tables: if t in font: del font[t] # 5. 保存 font.save(output_path) processed_paths.append(output_path) log_messages.append(f"✅ Success: {output_filename}") except Exception as e: log_messages.append(f"❌ Error: {filename} ({str(e)})") zip_path = f"fonts_fixed_{timestamp}.zip" with zipfile.ZipFile(zip_path, 'w') as zipf: for f in processed_paths: zipf.write(f, os.path.basename(f)) return zip_path, "\n".join(log_messages) # UI with gr.Blocks() as demo: gr.Markdown("### 📦 エラー回避重視・直接保存版") gr.Markdown("subsetリペアを使わず、直接テーブルを書き換えて保存します。") with gr.Row(): with gr.Column(): input_files = gr.File(label="フォント選択", file_count="multiple") sb_input = gr.Number(value=30, label="左右余白") btn = gr.Button("修正実行", variant="primary") with gr.Column(): status = gr.Textbox(label="ログ") output_zip = gr.File(label="ZIP") btn.click(fn=process_multiple_fonts_no_rename, inputs=[input_files, sb_input], outputs=[output_zip, status]) demo.launch()
